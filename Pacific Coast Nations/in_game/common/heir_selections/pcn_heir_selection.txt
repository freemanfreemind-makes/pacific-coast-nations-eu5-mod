#The dynasty is inherited from the mothers side, but both genders can rule
TRY_REPLACE:matrilineal_non_exclusive = {
	allows_noble_marriage = yes
	potential = {
		NOR = {
			succession_law = heir_selection:fratricide_succesion
			succession_law = heir_selection:unigeniture
		}
		OR = {
			culture.language = language:malayalam_language
			culture = culture:tuluva
			culture = culture:minangkabau_culture
			culture.language = language:kwa_language
			culture.language = language:nubian_language
			culture.language = language:olelo_hawaii_language
			culture.language = language:nadene_language
			culture.language = language:wakashan_language
			culture.language = language:tsimshian_language
			culture.language = language:haida_language
			culture.language = language:apsaalooke_hiraaca_language
			culture.language = language:hopi_language
			culture.language = language:zuni_language
			culture.language = language:keres_language
			culture.language = language:tanoan_language
			culture.language = language:lenape_language
			culture.language = language:powhatan_language
			culture.language = language:chibchan_language
			culture.language = language:bora_huitoto_language
			has_unlocked_heir_selection_trigger = { type = matrilineal_non_exclusive }
		}
	}
	allowed = {
		NOT = { modifier:disallows_female_rulers = yes }
	}
	through_female = yes
	allow_female = yes
	allow_male = yes
	ignore_ruler = yes
	allow_children = yes
	

	use_mothers_dynasty = yes
	all_in_country = yes

	allowed_estates = { nobles_estate clergy_estate burghers_estate}
	heir_is_allowed = {
		exists = root.dynasty
		scope:target.ruler_or_heir_if_regent ?= {
			dynasty ?= root.dynasty
		}
	}
	locked = {
	}

	succession_effect = {
		if = {
			limit = {
				government_type = government_type:steppe_horde
				modifier:horde_unity_hit_at_ruler_death < 0
			}
			custom_tooltip = ON_RULER_DEATH_EFFECT_TT
			show_as_tooltip = {
				add_horde_unity = root.modifier:horde_unity_hit_at_ruler_death
			}
		}
		add_political_influence = political_influence_mild_penalty
	}
	sibling_score = {
	}

	calc = {
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_ADM_DESC"
				value = root.adm
			}
			add = {
				desc = "HEIR_SELECTION_DIP_DESC"
				value = root.dip
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = root.mil
			}
			if = {
				limit = {
					has_dynasty = yes
					scope:target.ruler_or_regent ?= {
						has_dynasty = yes
						dynasty = prev.dynasty
					}
				}
				add = {
					desc = "HEIR_SELECTION_SAME_DYNASTY_AS_CURRENT_RULER"
					value = 100
				}
			}
			if = {
				limit = {
					OR = {
						exists = root.father
						exists = root.mother
					}
				}

				if = {
					limit = {
						any_parent = {
							is_ruler_of = scope:target
						}
					}
					add = {
						desc = "HEIR_SELECTION_PARENT_IS_CURRENT_RULER_DESC"
						value = 100
					}
				}
			}
			if = {
				limit = {
					OR = {
						is_general = yes
						is_admiral = yes
					}
				}
				add = {
					desc = "HEIR_SELECTION_GENERAL_OR_ADMIRAL_DESC"
					value = 50
				}
			}
			if = {
				limit = {
					OR = {
						has_trait_category = general
						has_trait_category = admiral
					}
				}
				add = {
					desc = "HEIR_SELECTION_MILITARY_TRAITS"
					value = 50
				}
			}
			if = {
				limit = {
					exists = scope:target.last_valid_ruler
					mother ?= scope:target.last_valid_ruler
				}
				add = {
					desc = "HEIR_SELECTION_MOTHER_IS_RULER_DESC"
					value = 500
				}
			}
			if = {
				limit = {
					is_female = yes
					exists = scope:target.last_valid_ruler
					exists = mother
					mother.mother ?= scope:target.last_valid_ruler
				}
				add = {
					desc = "HEIR_SELECTION_GRANDMOTHER_IS_RULER_DESC"
					value = 250
				}
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			add = {
				desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
				if = { limit = { modifier:blocked_from_being_ruler = yes } value = -1000 }
			}
		}
		else = {
			#description
			add = {
				desc = "HEIR_SELECTION_SAME_DYNASTY_AS_CURRENT_RULER"
				value = 100
			}
			add = {
				desc = "HEIR_SELECTION_PARENT_IS_CURRENT_RULER_DESC"
				value = 100
			}
			add = {
				desc = "HEIR_SELECTION_GENERAL_OR_ADMIRAL_DESC"
				value = 50
			}
			add = {
				desc = "HEIR_SELECTION_ADM_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_DIP_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_MIL_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_MILITARY_TRAITS"
				value = 50
			}
			add = {
				desc = "HEIR_SELECTION_MOTHER_IS_RULER_DESC"
				value = 500
			}
			add = {
				desc = "HEIR_SELECTION_GRANDMOTHER_IS_RULER_DESC"
				value = 250
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
		}
	}
}

TRY_REPLACE:tribal_matrilineal = {
	include_ruler_siblings = yes
	through_female = yes
	allow_female = yes
	allow_male = no
	allow_children = yes
	ignore_ruler = yes

	use_mothers_dynasty = yes

	allowed_estates = { nobles_estate }
	heir_is_allowed = {
	}

	potential = {
		NOR = {
			succession_law = heir_selection:fratricide_succesion
			succession_law = heir_selection:unigeniture
		}
		OR = {
			culture.language = language:haudenosaunee_language
			culture.language = language:iroquoian_language
			culture.language = language:wendat_language
			culture.language = language:skarureh_language
			has_reform = government_reform:haudenosaunee_clan_mothers
		}
	}

	succession_effect = {
		add_political_influence = political_influence_extreme_penalty
	}

	calc = {
		add = {
			desc = "HEIR_SELECTION_MOTHER_IS_RULER_DESC"
			if = {
				limit = {
					OR = {
						not = { exists = root }
						and = {
							exists = scope:target.last_valid_ruler
							mother ?= scope:target.last_valid_ruler
						}
					}
				}
				value = 500
			}
		}

		add = {
			desc = "HEIR_SELECTION_GRANDMOTHER_IS_RULER_DESC"
			if = {
				limit = {
					OR = {
						not = { exists = root }
						and = {
							exists = mother
							exists = scope:target.last_valid_ruler
							mother.mother ?= scope:target.last_valid_ruler
						}
					}
				}
				value = 250
			}
		}

		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			if = {
				limit = { modifier:blocked_from_being_ruler = yes }
				add = {
					desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
					value = -1000
				}
			}
		}

		# Here it goes the description for explaining a heir gets +1 points per year.
		if = {
			limit = {
				not = { exists = root }
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
		}
	}
}
